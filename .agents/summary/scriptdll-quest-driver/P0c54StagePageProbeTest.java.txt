package com.aionemu.gameserver.questEngine.retail;

import org.junit.jupiter.api.Test;

import java.io.InputStream;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Map;
import java.util.TreeMap;

/**
 * P0c-54 阶段页轴临时探针（用后即删、源码归档到 {@code .agents/summary/scriptdll-quest-driver/}）。
 * 打印受裁任务的**规范化 IR 文本**（{@link RetailIrFingerprint#canonicalText} 同口径），用于
 * 「剪行后原来那条页路由是否真的消失 / 入口补行是否真的进 IR」的逐任务机械对拍——
 * 指纹只在值变化时才可见，本探针给出**行级**证据（含未变化任务为何未变化）。
 * Temporary P0c-54 probe: dumps the canonical IR text per adjudicated quest so the change surface can
 * be diffed line by line against the pre-install registry.
 */
class P0c54StagePageProbeTest {

	@Test
	void dumpCanonicalIrForAdjudicatedQuests() throws Exception {
		String out = System.getProperty("p0c54.irOut");
		String ids = System.getProperty("p0c54.questIds");
		if (out == null || ids == null) {
			return;
		}
		RetailSimpleTalkTable table;
		try (InputStream input = open("/aion/data/static_data/quest_retail/Quest_SimpleTalk.xml")) {
			table = RetailSimpleTalkTable.load(input);
		}
		RetailQuestXmlTable retailTable;
		try (InputStream input = open("/aion/data/static_data/quest_retail/quest.xml")) {
			retailTable = RetailQuestXmlTable.load(input);
		}
		RetailClientDialogExits exits;
		try (InputStream input = open("/aion/data/static_data/quest_retail/quest_client_dialog_exits.tsv")) {
			exits = RetailClientDialogExits.load(input);
		}
		RetailClientSummaryRows summaryRows;
		try (InputStream input = open("/aion/data/static_data/quest_retail/quest_client_summary_rows.tsv")) {
			summaryRows = RetailClientSummaryRows.load(input);
		}
		RetailClientRewardNpcs clientRewardNpcs;
		try (InputStream input = open("/aion/data/static_data/quest_retail/quest_client_reward_npcs.tsv")) {
			clientRewardNpcs = RetailClientRewardNpcs.load(input);
		}
		RetailClientTalkChainSteps chainSteps;
		try (InputStream input = open("/aion/data/static_data/quest_retail/quest_client_talk_chain_steps.tsv")) {
			chainSteps = RetailClientTalkChainSteps.load(input);
		}
		RetailNpcNameIndex npcIndex = RetailNpcNameIndex.build(
			openAll("/aion/data/static_data/npcs/", List.of(
				"npc_template_200000_216188.xml", "npc_template_216189_235748.xml",
				"npc_template_235749_247606.xml", "npc_template_247607_270057.xml",
				"npc_template_270058_286320.xml", "npc_template_286321_800030.xml",
				"npc_template_800031_834289.xml", "npc_template_834290_885645.xml")),
			RetailQuestAiNameGroupsFixture.streams());
		RetailItemNameIndex itemIndex = RetailItemNameIndex.build(
			openAll("/aion/data/static_data/items/item/", listNames("/aion/data/static_data/items/item/")));
		Map<String, Integer> randomRewards = Fixtures.randomRewardIds();
		Map<Integer, Integer> nameIds = Fixtures.nameIds();

		StringBuilder text = new StringBuilder();
		for (String id : ids.split(",")) {
			int questId = Integer.parseInt(id.trim());
			var entry = table.find(questId);
			var meta = retailTable.find(questId);
			if (entry.isEmpty() || meta.isEmpty()) {
				text.append("### quest ").append(questId).append(": 输入缺失\n");
				continue;
			}
			var outcome = RetailSimpleTalkDefinitionCompiler.compile(entry.get(), npcIndex,
				RetailQuestMetadataCompiler.compile(meta.get(), npcIndex, itemIndex, randomRewards, nameIds),
				exits, summaryRows, clientRewardNpcs, chainSteps);
			text.append("### quest ").append(questId).append(" accepted=").append(outcome.accepted())
				.append(" rejection=").append(outcome.rejectionCode()).append('\n');
			if (!outcome.accepted()) {
				continue;
			}
			List<String> lines = new ArrayList<>(java.util.Arrays.asList(
				RetailIrFingerprint.canonicalText(outcome.definition().definition()).split("\n")));
			lines.sort(Comparator.naturalOrder());
			for (String line : lines) {
				text.append(line).append('\n');
			}
		}
		Path path = Path.of(out);
		Files.writeString(path, text.toString());
		System.out.println("P0C54-IR-DUMP quests=" + ids + " out=" + path + " chars=" + text.length());
	}

	private static InputStream open(String resource) {
		InputStream input = P0c54StagePageProbeTest.class.getResourceAsStream(resource);
		if (input == null) {
			throw new IllegalStateException("missing resource " + resource);
		}
		return input;
	}

	private static List<InputStream> openAll(String directory, List<String> names) {
		List<InputStream> streams = new ArrayList<>();
		for (String name : names) {
			streams.add(open(directory + name));
		}
		return streams;
	}

	private static List<String> listNames(String directory) throws Exception {
		var url = P0c54StagePageProbeTest.class.getResource(directory);
		if (url == null) {
			throw new IllegalStateException("missing directory " + directory);
		}
		List<String> names = new ArrayList<>();
		try (var paths = Files.list(Path.of(url.toURI()))) {
			paths.filter(path -> path.getFileName().toString().endsWith(".xml"))
				.map(path -> path.getFileName().toString()).sorted().forEach(names::add);
		}
		return names;
	}

	/** 与常设门禁同口径的读表助手（复制字面逻辑，避免探针与门禁口径分叉）。 */
	private static final class Fixtures {

		static Map<String, Integer> randomRewardIds() throws Exception {
			Map<String, Integer> ids = new java.util.HashMap<>();
			var factory = javax.xml.parsers.DocumentBuilderFactory.newInstance();
			factory.setFeature(javax.xml.XMLConstants.FEATURE_SECURE_PROCESSING, true);
			factory.setFeature("http://apache.org/xml/features/disallow-doctype-decl", true);
			var document = factory.newDocumentBuilder()
				.parse(open("/aion/data/static_data/quest_random_rewards.xml"));
			var nodes = document.getDocumentElement().getElementsByTagName("quest_random_reward");
			for (int index = 0; index < nodes.getLength(); index++) {
				var element = (org.w3c.dom.Element) nodes.item(index);
				ids.put(element.getElementsByTagName("name").item(0).getTextContent().trim(),
					Integer.parseInt(element.getElementsByTagName("id").item(0).getTextContent().trim()));
			}
			return ids;
		}

		static Map<Integer, Integer> nameIds() throws Exception {
			Map<Integer, Integer> ids = new java.util.HashMap<>();
			try (var reader = new java.io.BufferedReader(new java.io.InputStreamReader(
					open("/aion/data/static_data/quest_retail/quest_name_string_ids.tsv"),
					java.nio.charset.StandardCharsets.UTF_8))) {
				String line;
				while ((line = reader.readLine()) != null) {
					if (line.startsWith("#") || line.isBlank()) {
						continue;
					}
					String[] parts = line.split("\t");
					ids.put(Integer.parseInt(parts[0].substring("STR_QUEST_NAME_Q".length())),
						Integer.parseInt(parts[1]));
				}
			}
			return ids;
		}
	}
}
