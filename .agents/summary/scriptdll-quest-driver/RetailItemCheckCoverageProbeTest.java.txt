package com.aionemu.gameserver.questEngine.definition;

import com.aionemu.gameserver.questEngine.retail.RetailQuestDriver;
import com.aionemu.gameserver.questEngine.retail.RetailSimpleTalkTable;
import org.junit.jupiter.api.Test;

import java.io.BufferedReader;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.TreeSet;

/**
 * 临时探针（P0c-33）：adopted 真端行中声明 item_check 却缺 39/20002 门路由的行清单。
 * Temporary probe: adopted retail rows that declare item_check but emit no CHECK-button gate.
 */
class RetailItemCheckCoverageProbeTest {

	private static final int CHECK_USER_HAS_QUEST_ITEM = 39;
	private static final int CHECK_USER_HAS_QUEST_ITEM_SIMPLE = 20002;

	@Test
	void auditItemCheckCoverage() throws Exception {
		RetailSimpleTalkTable table;
		try (InputStream input = getClass().getResourceAsStream(
				"/aion/data/static_data/quest_retail/Quest_SimpleTalk.xml")) {
			table = RetailSimpleTalkTable.load(input);
		}
		Map<Integer, String> owners = retentionOwners();
		QuestCatalog catalog = RetailQuestDriver.overlay(
			QuestDefinitionDirectoryLoader.compile(getClass().getClassLoader()));
		List<String> missing = new ArrayList<>();
		List<String> xmlSide = new ArrayList<>();
		int present = 0;
		for (int questId : new TreeSet<>(table.questIds())) {
			RetailSimpleTalkTable.Entry entry = table.find(questId).orElseThrow();
			if (!entry.itemCheck()) {
				continue;
			}
			CompiledQuestDefinition compiled = catalog.findExecutable(questId).orElse(null);
			if (compiled == null) {
				continue;
			}
			long gates = compiled.definition().transitions().stream()
				.filter(transition -> transition.event() instanceof QuestEvent.TalkToNpc talk
					&& talk.dialogId() != null
					&& (talk.dialogId() == CHECK_USER_HAS_QUEST_ITEM
						|| talk.dialogId() == CHECK_USER_HAS_QUEST_ITEM_SIMPLE))
				.count();
			String owner = owners.getOrDefault(questId, "NO_ROW");
			if (gates > 0) {
				present++;
			} else if ("RETAIL_TABLE".equals(owner)) {
				long hundred9 = compiled.definition().transitions().stream()
					.filter(transition -> transition.event() instanceof QuestEvent.TalkToNpc talk
						&& talk.dialogId() != null && talk.dialogId() == 1009)
					.count();
				long hasItem = compiled.definition().transitions().stream()
					.filter(transition -> transition.conditions().stream()
						.anyMatch(QuestCondition.HasItem.class::isInstance))
					.count();
				missing.add(questId + "|talk=" + entry.talkNpcs() + "|rewardNpc=" + entry.rewardNpc()
					+ "|single=" + entry.singleStep() + "|route1009=" + hundred9 + "|hasItemEdges=" + hasItem);
			} else {
				xmlSide.add(questId + "|" + owner);
			}
		}
		System.out.println("ITEM_CHECK adopted missing=" + missing.size() + " present=" + present
			+ " nonAdoptedMissing=" + xmlSide.size());
		System.out.println("--- ADOPTED MISSING ---");
		missing.forEach(row -> System.out.println("  " + row));
	}

	private static Map<Integer, String> retentionOwners() throws Exception {
		Map<Integer, String> owners = new HashMap<>();
		try (InputStream input = RetailItemCheckCoverageProbeTest.class
				.getResourceAsStream("/quest/retail-xml-retention.tsv")) {
			BufferedReader reader = new BufferedReader(new InputStreamReader(input, StandardCharsets.UTF_8));
			String line;
			while ((line = reader.readLine()) != null) {
				if (line.isBlank() || line.startsWith("#")) {
					continue;
				}
				String[] fields = line.split("\t", -1);
				owners.put(Integer.parseInt(fields[0]), fields[1]);
			}
		}
		return owners;
	}
}
