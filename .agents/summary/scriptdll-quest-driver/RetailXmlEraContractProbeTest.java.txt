package com.aionemu.gameserver.questEngine.definition;

import org.junit.jupiter.api.Test;

import java.io.ByteArrayInputStream;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.List;
import java.util.Map;

/**
 * 临时探针（P0c-33）：24202/80320 页面 2375 审计差异的逐行归因——打印两期全部审计行与
 * started 态对话路由，定位遗留 XML 有、真端编译缺的那条边。
 * Temporary probe: per-row audit attribution plus started-state dialog routes for both eras.
 */
class RetailXmlEraContractProbeTest {

	private static final Path LEGACY_DIR = Path.of("/tmp/p0c33");

	@Test
	void auditLegacyXmlDefinitions() throws Exception {
		Map<Integer, QuestDialogOrderAudit.ClientQuest> clientQuests = QuestDialogOrderAudit.readClientPages(
			Path.of("docs/quest/client-dialog-mapping/quest-dialog-pages.csv"),
			Path.of("docs/quest/client-dialog-mapping/quest-dialog-action-details.csv"));
		QuestCatalog retail = com.aionemu.gameserver.questEngine.retail.RetailQuestDriver.overlay(
			QuestDefinitionDirectoryLoader.compile(getClass().getClassLoader()));
		for (int questId : new int[] {24202, 80320}) {
			dump(questId, "XML", compileLegacy(questId), clientQuests);
			dump(questId, "RETAIL", retail.findExecutable(questId).orElseThrow(), clientQuests);
		}
	}

	private static CompiledQuestDefinition compileLegacy(int questId) throws Exception {
		String text = Files.readString(LEGACY_DIR.resolve("legacy-" + questId + ".xml"),
			StandardCharsets.UTF_8);
		return QuestDefinitionXmlCompiler.compile(
			new ByteArrayInputStream(text.getBytes(StandardCharsets.UTF_8)));
	}

	private static void dump(int questId, String era, CompiledQuestDefinition compiled,
			Map<Integer, QuestDialogOrderAudit.ClientQuest> clientQuests) {
		QuestCatalog catalog = ImmutableQuestCatalog.fromEntries(
			List.of(QuestCatalogEntry.executable(compiled)));
		List<QuestDialogOrderAudit.AuditRow> rows = QuestDialogOrderAudit.audit(catalog, clientQuests);
		System.out.println("=== " + era + " quest " + questId + " auditRows=" + rows.size());
		rows.stream().filter(row -> "2375".equals(row.shownPage()))
			.forEach(row -> System.out.println("  PAGE2375 status=" + row.auditStatus() + " state="
				+ row.serverSourceState() + " npc=" + row.npcId() + " trigger=" + row.triggerAction()
				+ " clientAction=" + row.clientVisibleAction() + " candidates=" + row.candidateCount()
				+ " reason=" + (row.unresolvedReason().isEmpty() ? "-" : row.unresolvedReason())));
		QuestDefinition definition = compiled.definition();
		System.out.println("  -- dialog routes --");
		definition.transitions().stream()
			.filter(transition -> transition.event() instanceof QuestEvent.TalkToNpc)
			.forEach(transition -> {
				QuestEvent.TalkToNpc talk = (QuestEvent.TalkToNpc) transition.event();
				System.out.println("   " + transition.sourceNode() + " + npc " + talk.npcId() + " + action "
					+ talk.dialogId() + " -> " + transition.targetNode() + " after="
					+ transition.afterCommit());
			});
	}
}
