package com.aionemu.gameserver.questEngine.retail;

import com.aionemu.gameserver.questEngine.definition.ProductionQuestDefinitions;
import com.aionemu.gameserver.questEngine.definition.CompiledQuestDefinition;
import com.aionemu.gameserver.questEngine.definition.QuestNode;
import com.aionemu.gameserver.questEngine.definition.QuestTransition;
import com.aionemu.gameserver.questEngine.model.QuestStatus;
import org.junit.jupiter.api.Test;

import java.io.PrintWriter;
import java.util.List;
import java.util.Set;
import java.util.stream.Collectors;

import static org.junit.jupiter.api.Assertions.assertNotNull;

/**
 * 临时探针（P0c-14）：ReportToMany 批 16 行的真端编译接取路由直证，结果 -Dretail.talkReport.out 导出。
 * Temporary probe for the ReportToMany batch: start-dialog route presence in the retail compile.
 */
class RetailTalkReportRouteProbeTest {

	private static final List<String> ROUTES = List.of(
		"80268|831172|2375", "80270|831178|2375", "80310|831384|1003", "80311|831387|1003",
		"80324|831447|2375", "80325|831447|2375", "80326|831448|2375", "80327|831448|2375",
		"80353|831818|2375", "80354|831818|2375", "80355|831818|2375", "80362|831830|2375",
		"80363|831830|2375", "80364|831830|2375", "80623|832870|2375", "80624|832871|2375");

	@Test
	void dumpStartDialogRoutes() throws Exception {
		StringBuilder out = new StringBuilder();
		for (String spec : ROUTES) {
			String[] parts = spec.split("\\|");
			int questId = Integer.parseInt(parts[0]);
			int npcId = Integer.parseInt(parts[1]);
			int pageId = Integer.parseInt(parts[2]);
			CompiledQuestDefinition compiled = ProductionQuestDefinitions.definitionInOverlay(questId);
			assertNotNull(compiled, "no retail compile " + questId);
			var definition = compiled.definition();
			Set<String> startedNodes = definition.nodes().stream()
				.filter(node -> node.projection().status() == QuestStatus.START)
				.map(QuestNode::label)
				.collect(Collectors.toSet());
			boolean has = definition.transitions().stream().anyMatch(transition ->
				startedNodes.contains(transition.sourceNode())
					&& transition.event() instanceof com.aionemu.gameserver.questEngine.definition.QuestEvent.TalkToNpc talk
					&& talk.npcId() == npcId
					&& Integer.valueOf(31).equals(talk.dialogId())
					&& transition.afterCommit().contains(
						new com.aionemu.gameserver.questEngine.definition.AfterCommitAction.ShowQuestDialog(pageId)));
			out.append(questId).append('\t').append(has ? "ROUTE_OK" : "ROUTE_MISSING").append('\n');
		}
		String path = System.getProperty("retail.talkReport.out");
		if (path != null) {
			try (PrintWriter writer = new PrintWriter(path, "utf-8")) {
				writer.write(out.toString());
			}
		}
		System.out.println(out);
		for (String line : out.toString().split("\n")) {
			assertNotNull(line, line);
		}
	}
}
