package com.aionemu.gameserver.questEngine.definition;

import com.aionemu.gameserver.questEngine.model.QuestStatus;
import com.aionemu.gameserver.questEngine.retail.RetailQuestDriver;

import org.junit.jupiter.api.Test;

import java.util.List;

/**
 * 临时探针（P0c-35）：本轮入台行（24202/80320/1152）的 EnterWorld 领奖行修复边形状——
 * JournalRewardRowRepairContractTest 因 15613（DataDriven 车道行）在列表更早处中止，24202 断言被掩蔽。
 * Temporary probe: the reward-row recovery edge shape for this slice's rows.
 */
class P0c35RewardRecoveryEdgeProbeTest {

	@Test
	void rewardRecoveryEdgesForThisSlice() throws Exception {
		QuestCatalog catalog = RetailQuestDriver.overlay(
			QuestDefinitionDirectoryLoader.compile(getClass().getClassLoader()));
		for (int questId : new int[] {24202, 80320, 1152}) {
			QuestDefinition definition = catalog.findExecutable(questId)
				.orElseThrow(() -> new IllegalStateException("missing " + questId)).definition();
			List<QuestTransition> edges = definition.transitions().stream()
				.filter(transition -> transition.event() instanceof QuestEvent.EnterWorld).toList();
			System.out.println("=== quest " + questId + " enterWorldEdges=" + edges.size());
			edges.forEach(edge -> System.out.println("   " + edge.targetNode() + " cond="
				+ edge.conditions() + " act=" + edge.actions() + " after=" + edge.afterCommit()));
			definition.nodes().stream()
				.filter(node -> node.projection().status() == QuestStatus.START
					|| node.projection().status() == QuestStatus.REWARD)
				.forEach(node -> System.out.println("   node " + node.label() + " "
					+ node.projection().status() + " var0="
					+ node.projection().variables().get("var0")));
			System.out.println("   drops=" + definition.metadata().drops().stream()
				.map(drop -> drop.npcId() + ":" + drop.itemId() + ":step" + drop.collectingStep()).toList());
		}
	}
}
