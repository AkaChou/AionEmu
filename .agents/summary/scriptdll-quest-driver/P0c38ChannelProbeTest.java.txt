package com.aionemu.gameserver.questEngine.definition;

import com.aionemu.gameserver.questEngine.retail.RetailQuestDriver;

import org.junit.jupiter.api.Test;

import java.nio.file.Path;
import java.util.List;
import java.util.Map;

/**
 * 临时探针（P0c-38）：①P0c-38 新合成的 10 行（XML_RETENTION）在真端编译器下的实时判定 + 生产视图
 * 的节点/路由/页面可达性（验证"合成记录对 XML 拥有行是惰性的"）；②符号通道修正后的 2458/4905/4906
 * 的物品动作（give/remove）逐字，用于与 p0c38-canonical-item-channel.tsv 的旧值对拍。
 * Temporary P0c-38 probe: live retail-compile verdicts, production-view reachability, item actions.
 */
class P0c38ChannelProbeTest {

	/** P0c-38 新合成行（真端索引通道）。 */
	private static final int[] SYNTHESIZED = {18035, 18807, 18809, 21070, 21460, 24120, 28035, 28807, 29070, 29071};

	/** 符号通道分歧行（旧序数通道给的值 vs 真端名索引）。 */
	private static final int[] DIVERGED = {2458, 4905, 4906};

	@Test
	void synthesizedRowsAndItemChannel() throws Exception {
		QuestCatalog catalog = RetailQuestDriver.overlay(
			QuestDefinitionDirectoryLoader.compile(getClass().getClassLoader()));
		Map<Integer, QuestDialogOrderAudit.ClientQuest> clientQuests = QuestDialogOrderAudit.readClientPages(
			Path.of("docs/quest/client-dialog-mapping/quest-dialog-pages.csv"),
			Path.of("docs/quest/client-dialog-mapping/quest-dialog-action-details.csv"));
		List<QuestDialogOrderAudit.AuditRow> rows = QuestDialogOrderAudit.audit(catalog, clientQuests);
		for (int questId : SYNTHESIZED) {
			var compiled = catalog.findExecutable(questId).orElse(null);
			System.out.println("=== " + questId + " executable=" + (compiled != null));
			if (compiled != null) {
				QuestDefinition definition = compiled.definition();
				System.out.println("  nodes=" + definition.nodes().stream()
					.map(node -> node.label() + ":" + node.projection().status()
						+ node.projection().variables()).toList());
				System.out.println("  transitions=" + definition.transitions().size()
					+ " drops=" + definition.metadata().drops());
			}
			rows.stream().filter(row -> row.questId() == questId).forEach(row ->
				System.out.println("  AUDIT page=" + row.shownPage() + " client=" + row.clientVisibleAction()
					+ " status=" + row.auditStatus()));
		}
		for (int questId : DIVERGED) {
			var compiled = catalog.findExecutable(questId).orElse(null);
			System.out.println("=== " + questId + " item actions");
			if (compiled == null) {
				System.out.println("  (no executable definition)");
				continue;
			}
			compiled.definition().transitions().forEach(transition -> {
				if (transition.actions().isEmpty() && transition.afterCommit().isEmpty()) {
					return;
				}
				System.out.println("  T " + transition.sourceNode() + "->" + transition.targetNode()
					+ " ev=" + transition.event() + " act=" + transition.actions()
					+ " after=" + transition.afterCommit());
			});
		}
	}
}
