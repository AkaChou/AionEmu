/**
 * 临时探针（P0c-29）：宽松窗口回放——风暴期以 lenient overlay 逐任务回放 P0c-24~28 全部
 * 待窗判官断言（本 lane 各任务均为真端 owner 且可编译，全量 fail-closed 门是唯一被阻项）。
 * 绿 = 窗口收窄为"正式判官复跑 + T2/T3 门禁"；红 = 翻转质量问题即刻暴露。
 * Temporary probe: replays every pending window judge's assertions over the lenient overlay
 * (all our quests are retail-owned and compile; only the fail-closed full-view gate is blocked).
 * Green shrinks the real window to official judge re-runs plus gates.
 */
class RetailLenientWindowReplayProbeTest {

	@Test
	void daevanionArmourHealEdgesAndProjections() {
		for (int questId : new int[] {80290, 80294}) {
			QuestDefinition definition = overlay(questId);
			// 自愈边：唯一 EnterWorld 恢复路由，[REWARD,var0=1]→var0:=0 + SyncQuestState。
			List<QuestTransition> heals = definition.transitions().stream()
				.filter(transition -> transition.sourceNode() == null
					&& "reward".equals(transition.targetNode())
					&& transition.event().equals(new QuestEvent.EnterWorld()))
				.toList();
			assertEquals(1, heals.size(), () -> questId + " heal route count");
			QuestTransition heal = heals.getFirst();
			assertEquals(List.of(new QuestCondition.StatusIs(QuestStatus.REWARD),
				new QuestCondition.QuestVariableIs("var0", 1)), heal.conditions(),
				() -> questId + " heal conditions");
			assertEquals(List.of(new QuestAction.SetVariable("var0", 0)), heal.actions(),
				() -> questId + " heal actions");
			assertEquals(List.of(new AfterCommitAction.SyncQuestState(
				QuestStateSyncMode.LEVEL_AND_VISIBILITY_REFRESH)), heal.afterCommit(),
				() -> questId + " heal after-commit");
			// armour 投影：reward 节点 var0=0；完成 owner = 家族事件 NPC。
			assertEquals(Map.of("var0", 0), node(definition, "reward").projection().variables(),
				() -> questId + " armour reward projection");
			assertEquals(QuestStatus.REWARD, node(definition, "reward").projection().status());
			assertEquals(Set.of(questId == 80290 ? 831384 : 831387), completionOwners(definition),
				() -> questId + " completion owner");
		}
	}

	@Test
	void quest1141BarrelReportShape() {
		QuestDefinition definition = overlay(1141);
		assertPage(definition, "unaccepted", 730001, QuestDialogAction.QUEST_SELECT,
			QuestDialogPage.SELECT1);
		assertPage(definition, "started", 700122, QuestDialogAction.QUEST_SELECT,
			QuestDialogPage.SELECT5);
		QuestTransition report = route(definition, "started", 700122,
			QuestDialogAction.SELECT_QUEST_REWARD);
		assertEquals("reward", report.targetNode());
		assertEquals(List.of(
			new AfterCommitAction.SyncQuestState(QuestStateSyncMode.LEVEL_AND_VISIBILITY_REFRESH),
			new AfterCommitAction.ShowQuestDialog(QuestDialogPage.SHOW_SELECT_QUEST_REWARD_WINDOW1.id())),
			report.afterCommit());
		assertEquals("complete", route(definition, "reward", 700122,
			QuestDialogAction.SELECTED_QUEST_REWARD1).targetNode());
		// 无掉落 → 无交互门（P0c-24 掉落驱动判据）。
		assertTrue(definition.transitions().stream()
			.noneMatch(transition -> transition.event() instanceof QuestEvent.CanAct canAct
				&& canAct.templateId() == 700122), "1141 must not carry an object gate");
	}

	@Test
	void quests30312And30315GroupDropsAndTwoRouteTurnIns() {
		QuestDefinition d30312 = overlay(30312);
		assertEquals(Set.of(
			new QuestDrop(216007, 182209715, 50, true, 0),
			new QuestDrop(216008, 182209715, 50, true, 0)
		), Set.copyOf(d30312.metadata().drops()));
		assertTwoRouteTurnIns(d30312, 182209715, 20, null);

		QuestDefinition d30315 = overlay(30315);
		assertTwoRouteTurnIns(d30315, 186000098, 200, 799226);
	}

	private void assertTwoRouteTurnIns(QuestDefinition definition, int itemId, int count, Integer npcFilter) {
		List<QuestTransition> turnIns = definition.transitions().stream()
			.filter(transition -> "started".equals(transition.sourceNode()))
			.filter(transition -> "reward".equals(transition.targetNode()))
			.filter(transition -> transition.event() instanceof QuestEvent.TalkToNpc talk
				&& (npcFilter == null || talk.npcId() == npcFilter))
			.toList();
		assertEquals(2, turnIns.size(), "two family turn-in routes");
		assertEquals(Set.of(39, 20002), turnIns.stream()
			.map(transition -> ((QuestEvent.TalkToNpc) transition.event()).dialogId())
			.collect(Collectors.toSet()));
		for (QuestTransition turnIn : turnIns) {
			assertEquals(List.of(new QuestCondition.HasItem(itemId, count)), turnIn.conditions());
			assertEquals(List.of(new QuestAction.RemoveItem(itemId, count)), turnIn.actions());
		}
	}

	@Test
	void quest1351FamilyShapeIncludingTheAbortMaskedTurnInBlock() {
		QuestDefinition definition = overlay(1351);
		// 接取 NPC：族形 started 仅 FINISH_DIALOG 出口 → 任务接取页；started 无 QUEST_SELECT
		// 描述页（unaccepted 的 QUEST_SELECT 接取流是合法族形——探针首跑证伪了我方 P0c-27 判官
		// 的过宽断言并已同步修正）；不拥有交付路由（P0c-27 判官重写）。
		assertFalse(definition.transitions().stream().anyMatch(candidate ->
				candidate.event() instanceof QuestEvent.TalkToNpc talk && talk.npcId() == 203965
					&& "started".equals(candidate.sourceNode())
					&& talk.dialogId() == QuestDialogAction.QUEST_SELECT.id()),
			"203965 must not open a QUEST_SELECT description page in the started state");
		QuestTransition exit = transition(definition, "started", "started",
			new QuestEvent.TalkToNpc(203965, QuestDialogAction.FINISH_DIALOG.id()));
		assertEquals(List.of(new AfterCommitAction.ShowQuestSelectionDialog(
			QuestDialogPage.SELECT_QUEST.id())), exit.afterCommit());
		assertTrue(definition.transitions().stream()
			.noneMatch(candidate -> candidate.event() instanceof QuestEvent.TalkToNpc talk
				&& talk.npcId() == 203965 && "reward".equals(candidate.targetNode())),
			"203965 must not own reward routes");
		// 交付 NPC 203983：入口 QUEST_SELECT→SELECT5 + 39/20002 双分支（judge-abort 掩蔽块，
		// 真端侧首次验证）：prio-0 成功路 HasItem/RemoveItem+奖励窗、prio-1 失败路空条件→SELECT6。
		assertEquals(List.of(new AfterCommitAction.ShowQuestDialog(QuestDialogPage.SELECT5.id())),
			transition(definition, "started", "started",
				new QuestEvent.TalkToNpc(203983, QuestDialogAction.QUEST_SELECT.id())).afterCommit());
		QuestTransition success = transition(definition, "started", "reward",
			new QuestEvent.TalkToNpc(203983, QuestDialogAction.CHECK_USER_HAS_QUEST_ITEM.id()));
		assertEquals(Integer.valueOf(0), success.priority());
		assertEquals(List.of(new QuestCondition.HasItem(182201321, 10)), success.conditions());
		assertEquals(List.of(new QuestAction.RemoveItem(182201321, 10)), success.actions());
		assertEquals(List.of(
			new AfterCommitAction.SyncQuestState(QuestStateSyncMode.LEVEL_AND_VISIBILITY_REFRESH),
			new AfterCommitAction.ShowQuestDialog(QuestDialogPage.SHOW_SELECT_QUEST_REWARD_WINDOW1.id())),
			success.afterCommit());
		QuestTransition failure = transition(definition, "started", "started",
			new QuestEvent.TalkToNpc(203983, QuestDialogAction.CHECK_USER_HAS_QUEST_ITEM.id()));
		assertEquals(Integer.valueOf(1), failure.priority());
		assertEquals(List.of(), failure.conditions());
		assertEquals(List.of(new AfterCommitAction.ShowQuestDialog(QuestDialogPage.SELECT6.id())),
			failure.afterCommit());
	}

	@Test
	void quest1526AlignedRewardRow() {
		assertEquals(Map.of("var0", 1), node(overlay(1526), "reward").projection().variables(),
			"1526 projection must equal the aligned sibling row (summary_rows 2 -> last index 1)");
	}

	private QuestDefinition overlay(int questId) {
		return ProductionQuestDefinitions.definitionInOverlay(questId).definition();
	}

	private QuestNode node(QuestDefinition definition, String label) {
		return definition.nodes().stream().filter(candidate -> label.equals(candidate.label()))
			.findFirst().orElseThrow(() -> new AssertionError("missing node " + label));
	}

	private Set<Integer> completionOwners(QuestDefinition definition) {
		return definition.transitions().stream()
			.filter(transition -> "complete".equals(transition.targetNode())
				&& transition.event() instanceof QuestEvent.TalkToNpc talk)
			.map(transition -> ((QuestEvent.TalkToNpc) transition.event()).npcId())
			.collect(Collectors.toSet());
	}

	private void assertPage(QuestDefinition definition, String source, int npcId,
			QuestDialogAction action, QuestDialogPage page) {
		QuestTransition transition = route(definition, source, npcId, action);
		assertEquals(source, transition.targetNode());
		assertEquals(List.of(new AfterCommitAction.ShowQuestDialog(page.id())), transition.afterCommit());
	}

	private QuestTransition route(QuestDefinition definition, String source, int npcId,
			QuestDialogAction action) {
		return definition.transitions().stream()
			.filter(transition -> transition.sourceNode().equals(source)
				&& transition.event().equals(new QuestEvent.TalkToNpc(npcId, action.id())))
			.findFirst().orElseThrow();
	}

	private QuestTransition transition(QuestDefinition definition, String source, String target,
			QuestEvent event) {
		return definition.transitions().stream()
			.filter(candidate -> Objects.equals(candidate.sourceNode(), source)
				&& candidate.targetNode().equals(target) && candidate.event().equals(event))
			.findFirst().orElseThrow();
	}
}
