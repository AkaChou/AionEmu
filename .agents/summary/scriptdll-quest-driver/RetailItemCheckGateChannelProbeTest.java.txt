	@Test
	void threeItemReportRowsCompileAndPassTheContractAudit() throws Exception {
		Map<Integer, QuestDialogOrderAudit.ClientQuest> clientQuests = QuestDialogOrderAudit.readClientPages(
			Path.of("docs/quest/client-dialog-mapping/quest-dialog-pages.csv"),
			Path.of("docs/quest/client-dialog-mapping/quest-dialog-action-details.csv"));
		for (int questId : new int[] {24202, 80320, 1152}) {
			var entry = table.find(questId);
			var meta = retailTable.find(questId);
			System.out.println("=== quest " + questId + " itemReports="
				+ chainSteps.itemReports(questId).size() + " entry=" + entry.isPresent()
				+ " meta=" + meta.isPresent());
			if (entry.isEmpty() || meta.isEmpty()) {
				continue;
			}
			var outcome = RetailSimpleTalkDefinitionCompiler.compile(entry.get(), npcIndex,
				RetailQuestMetadataCompiler.compile(meta.get(), npcIndex, itemIndex, randomRewards, nameIds),
				exits, summaryRows, clientRewardNpcs, chainSteps);
			if (!outcome.accepted()) {
				System.out.println("  REJECTED " + outcome.rejectionCode() + " " + outcome.detail());
				continue;
			}
			QuestDefinition definition = outcome.definition().definition();
			System.out.println("  nodes=" + definition.nodes().size() + " transitions="
				+ definition.transitions().size());
			for (QuestTransition transition : definition.transitions()) {
				if (!(transition.event() instanceof QuestEvent.TalkToNpc talk)
						|| talk.dialogId() == null
						|| (talk.dialogId() != 39 && talk.dialogId() != 20002)) {
					continue;
				}
				System.out.println("  GATE " + transition.sourceNode() + " + npc " + talk.npcId()
					+ " + action " + talk.dialogId() + " -> " + transition.targetNode()
					+ " prio=" + transition.priority() + " cond=" + transition.conditions()
					+ " act=" + transition.actions() + " after=" + transition.afterCommit());
			}
			if (questId != 24202) {
				Path xml = Path.of("src/main/resources/aion/data/static_data/quest_definition/quests/"
					+ questId + ".xml");
				if (java.nio.file.Files.exists(xml)) {
					CompiledQuestDefinition legacy = QuestDefinitionXmlCompiler.compile(
						new java.io.ByteArrayInputStream(
							java.nio.file.Files.readString(xml).getBytes(java.nio.charset.StandardCharsets.UTF_8)));
					java.util.Set<String> retailKeys = keys(definition);
					java.util.Set<String> xmlKeys = keys(legacy.definition());
					java.util.Set<String> onlyRetail = new java.util.TreeSet<>(retailKeys);
					onlyRetail.removeAll(xmlKeys);
					java.util.Set<String> onlyXml = new java.util.TreeSet<>(xmlKeys);
					onlyXml.removeAll(retailKeys);
					System.out.println("  XML-EQUIV onlyRetail=" + onlyRetail.size()
						+ " onlyXml=" + onlyXml.size());
					onlyRetail.forEach(key -> System.out.println("   RETAIL_ONLY " + key));
					onlyXml.forEach(key -> System.out.println("   XML_ONLY " + key));
				}
			}
			QuestCatalog catalog = ImmutableQuestCatalog.fromEntries(
				List.of(QuestCatalogEntry.executable(outcome.definition())));
			List<QuestDialogOrderAudit.AuditRow> rows = QuestDialogOrderAudit.audit(catalog, clientQuests);
			System.out.println("  auditRows=" + rows.size());
			rows.forEach(row -> System.out.println("   AUDIT page=" + row.shownPage() + " client="
				+ row.clientVisibleAction() + " status=" + row.auditStatus() + " reason="
				+ row.unresolvedReason()));
		}
	}

	private static java.util.Set<String> keys(QuestDefinition definition) {
		java.util.Set<String> keys = new java.util.TreeSet<>();
		for (QuestTransition transition : definition.transitions()) {
			keys.add(transition.sourceNode() + "|" + transition.event() + "|" + transition.targetNode()
				+ "|" + transition.conditions() + "|" + transition.actions() + "|"
				+ transition.afterCommit() + "|" + transition.priority());
		}
		return keys;
	}

	private static Map<String, Integer> randomRewardIds() throws Exception {
		Map<String, Integer> ids = new HashMap<>();
		var document = parse(open("/aion/data/static_data/quest_random_rewards.xml"));
		var nodes = document.getDocumentElement().getElementsByTagName("quest_random_reward");
		for (int index = 0; index < nodes.getLength(); index++) {
			var element = (org.w3c.dom.Element) nodes.item(index);
			ids.put(element.getElementsByTagName("name").item(0).getTextContent().trim(),
				Integer.parseInt(element.getElementsByTagName("id").item(0).getTextContent().trim()));
		}
		return ids;
	}

	private static Map<Integer, Integer> nameIds() throws Exception {
		Map<Integer, Integer> ids = new HashMap<>();
		for (String line : lines(open("/aion/data/static_data/quest_retail/quest_name_string_ids.tsv"))) {
			if (line.startsWith("#") || line.isBlank()) {
				continue;
			}
			String[] parts = line.split("\t");
			ids.put(Integer.parseInt(parts[0].substring("STR_QUEST_NAME_Q".length())),
				Integer.parseInt(parts[1]));
		}
		return ids;
	}

	private static List<String> listXmlNames(String dir) throws Exception {
		var url = RetailItemCheckGateChannelProbeTest.class.getResource(dir);
		java.io.File[] files = new java.io.File(url.toURI()).listFiles((d, name) -> name.endsWith(".xml"));
		return java.util.Arrays.stream(files).map(java.io.File::getName).sorted().toList();
	}

	private static List<InputStream> openAll(String dir, List<String> files) throws Exception {
		List<InputStream> inputs = new ArrayList<>();
		for (String file : files) {
			inputs.add(open(dir + file));
		}
		return inputs;
	}

	private static org.w3c.dom.Document parse(InputStream input) throws Exception {
		try (input) {
			var factory = javax.xml.parsers.DocumentBuilderFactory.newInstance();
			factory.setFeature(javax.xml.XMLConstants.FEATURE_SECURE_PROCESSING, true);
			factory.setFeature("http://apache.org/xml/features/disallow-doctype-decl", true);
			return factory.newDocumentBuilder().parse(input);
		}
	}

	private static List<String> lines(InputStream input) throws Exception {
		try (var reader = new java.io.BufferedReader(
				new java.io.InputStreamReader(input, java.nio.charset.StandardCharsets.UTF_8))) {
			return reader.lines().toList();
		}
	}

	private static InputStream open(String resource) throws Exception {
		InputStream input = RetailItemCheckGateChannelProbeTest.class.getResourceAsStream(resource);
		if (input == null) {
			throw new IllegalStateException("missing resource " + resource);
		}
		return input;
	}

	/** 生产视图（XML + 真端 overlay）下的致命契约指纹，按任务过滤——与 QuestClientContractGateTest 同口径。 */
	@Test
	void productionFatalFingerprintsForTheFlippedRow() throws Exception {
		QuestCatalog catalog = com.aionemu.gameserver.questEngine.retail.RetailQuestDriver.overlay(
			QuestDefinitionDirectoryLoader.compile(getClass().getClassLoader()));
		Map<Integer, QuestDialogOrderAudit.ClientQuest> clientQuests = QuestDialogOrderAudit.readClientPages(
			Path.of("docs/quest/client-dialog-mapping/quest-dialog-pages.csv"),
			Path.of("docs/quest/client-dialog-mapping/quest-dialog-action-details.csv"));
		List<QuestDialogOrderAudit.AuditRow> rows = QuestDialogOrderAudit.audit(catalog, clientQuests);
		long fatal = rows.stream().filter(row -> row.unresolvedReason()
			.startsWith("visible client action has no route")
			|| row.unresolvedReason().startsWith("compiled IR emits a task page absent from the active client page index"))
			.count();
		System.out.println("PRODUCTION-FATAL total=" + fatal);
		rows.stream().filter(row -> row.unresolvedReason()
			.startsWith("visible client action has no route")
			|| row.unresolvedReason().startsWith("compiled IR emits a task page absent from the active client page index"))
			.filter(row -> row.questId() == 1152 || row.questId() == 24202 || row.questId() == 80320)
			.forEach(row -> System.out.println("  FATAL " + row.questId() + " state=" + row.serverSourceState()
				+ " npc=" + row.npcId() + " trigger=" + row.triggerAction() + " page=" + row.shownPage()
				+ " clientAction=" + row.clientVisibleAction() + " reason=" + row.unresolvedReason()
				+ " path=" + row.actualPath()));
		List<QuestDialogOrderAudit.AuditRow> mine = rows.stream()
			.filter(row -> row.questId() == 1152).toList();
		System.out.println("1152 audit rows=" + mine.size());
		mine.forEach(row -> System.out.println("   R1152 page=" + row.shownPage() + " client="
			+ row.clientVisibleAction() + " status=" + row.auditStatus() + " reason="
			+ row.unresolvedReason()));
	}
