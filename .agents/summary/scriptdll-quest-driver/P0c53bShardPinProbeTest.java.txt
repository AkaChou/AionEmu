package com.aionemu.gameserver.questEngine.retail;

import com.aionemu.gameserver.questEngine.definition.CompiledQuestDefinition;
import com.aionemu.gameserver.questEngine.definition.ProductionQuestDefinitions;
import com.aionemu.gameserver.questEngine.definition.QuestEvent;
import com.aionemu.gameserver.questEngine.definition.QuestTransition;

import org.junit.jupiter.api.Test;

import java.io.PrintStream;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.TreeMap;

/**
 * 临时探针（P0c-53b）：50073/50074 退役后生产 IR 的形状逐条取证，用于重定事件碎片 pin。
 * Temporary P0c-53b probe: dumps the production IR shape of the retired 50073/50074 rows.
 */
class P0c53bShardPinProbeTest {

	@Test
	void dumpsShardRows() throws Exception {
		Path out = Path.of(System.getProperty("p0c53b.probeOut"));
		try (PrintStream stream = new PrintStream(Files.newOutputStream(out), true, StandardCharsets.UTF_8)) {
			for (int questId : new int[] {50073, 50074, 13758, 15571, 13841}) {
				CompiledQuestDefinition compiled = ProductionQuestDefinitions.definitionInOverlay(questId);
				var metadata = compiled.definition().metadata();
				stream.println("== " + questId + " name=" + metadata.name()
					+ " displayNameId=" + metadata.displayNameId() + " minLevel=" + metadata.minLevel()
					+ " category=" + metadata.category() + " races=" + metadata.permittedRaces()
					+ " repeat=" + metadata.repeatPolicy() + " rewards=" + metadata.rewards());
				var var1 = compiled.definition().progressLayout().field("var1");
				stream.println("   var1=" + (var1 == null ? "null" : "max=" + var1.maxValue()));
				Map<String, Integer> grouped = new TreeMap<>();
				Map<String, Integer> killTargets = new TreeMap<>();
				for (QuestTransition transition : compiled.definition().transitions()) {
					String key = transition.sourceNode() + " -> " + transition.targetNode()
						+ " | " + transition.event().getClass().getSimpleName();
					grouped.merge(key, 1, Integer::sum);
					if (transition.event() instanceof QuestEvent.KillNpcSet set) {
						killTargets.merge("KillNpcSet " + set.npcIds(), 1, Integer::sum);
					} else if (transition.event() instanceof QuestEvent.KillNpc kill) {
						killTargets.merge("KillNpc " + kill.npcId(), 1, Integer::sum);
					} else if (transition.event() instanceof QuestEvent.KillInWorld world) {
						killTargets.merge("KillInWorld " + world.worldId(), 1, Integer::sum);
					}
				}
				stream.println("   kill events: " + killTargets);
				stream.println("   progress fields: " + compiled.definition().progressLayout());
				if (questId > 50000) {
					Map<String, Integer> completion = new LinkedHashMap<>();
					for (QuestTransition transition : compiled.definition().transitions()) {
						if ("reward".equals(transition.sourceNode()) && "complete".equals(transition.targetNode())
							&& transition.event() instanceof QuestEvent.TalkToNpc talk) {
							completion.merge("npcs=" + talk.npcId() + " dialog=" + talk.dialogId()
								+ " actions=" + transition.actions(), 1, Integer::sum);
						}
					}
					stream.println("   reward->complete routes:");
					completion.forEach((key, count) -> stream.println("     " + count + " × " + key));
				}
				stream.println("   transition shape:");
				grouped.forEach((key, count) -> stream.println("     " + count + " × " + key));
			}
		}
	}
}
