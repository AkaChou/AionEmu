package com.aionemu.gameserver.questEngine.definition;

import com.aionemu.gameserver.questEngine.model.QuestStatus;
import com.aionemu.gameserver.questEngine.retail.RetailQuestDriver;
import org.junit.jupiter.api.Test;

/**
 * 临时探针（P0c-33）：打印 24202 / 80320 / 1323 等窗口行的 EnterWorld 边、节点投影与页面路由，
 * 用于窗口期测试失败的归因（回退 or 对齐）。
 * Temporary probe: dump the window-era rows' EnterWorld edges and node projections for attribution.
 */
class RetailWindowRowShapeProbeTest {

	@Test
	void dumpWindowRowShapes() {
		QuestCatalog catalog = RetailQuestDriver.overlay(
			QuestDefinitionDirectoryLoader.compile(getClass().getClassLoader()));
		for (int questId : new int[] {24202, 80320, 1323, 1479, 2611, 3001, 3023, 21136, 24123}) {
			QuestDefinition definition = catalog.find(questId).orElseThrow().definition();
			System.out.println("=== quest " + questId + " nodes=" + definition.nodes().size()
				+ " transitions=" + definition.transitions().size());
			for (QuestNode node : definition.nodes()) {
				System.out.println("  node " + node.label() + " status=" + node.projection().status()
					+ " vars=" + node.projection().variables());
			}
			for (QuestTransition transition : definition.transitions()) {
				if (transition.event() instanceof QuestEvent.EnterWorld) {
					System.out.println("  EW src=" + transition.sourceNode() + " tgt=" + transition.targetNode()
						+ " conds=" + transition.conditions() + " actions=" + transition.actions()
						+ " after=" + transition.afterCommit() + " prio=" + transition.priority());
				}
			}
			long started2375 = definition.transitions().stream()
				.filter(t -> "started".equals(t.sourceNode()) && t.event() instanceof QuestEvent.TalkToNpc talk
					&& talk.dialogId() != null && talk.dialogId() == 31)
				.count();
			System.out.println("  started+action31 routes=" + started2375);
			definition.transitions().stream()
				.filter(t -> t.event() instanceof QuestEvent.TalkToNpc)
				.filter(t -> "started".equals(t.sourceNode()))
				.forEach(t -> {
					QuestEvent.TalkToNpc talk = (QuestEvent.TalkToNpc) t.event();
					System.out.println("   started talk npc=" + talk.npcId() + " action=" + talk.dialogId()
						+ " -> " + t.targetNode());
				});
			System.out.println("  rewardStatus=" + definition.nodes().stream()
				.filter(n -> "reward".equals(n.label())).findFirst()
				.map(n -> n.projection().status() == QuestStatus.REWARD).orElse(null));
		}
	}
}
