package com.aionemu.gameserver.questEngine.definition;

import org.junit.jupiter.api.Test;

import java.io.BufferedReader;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.nio.charset.StandardCharsets;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.LinkedHashSet;
import java.util.List;
import java.util.Map;
import java.util.Optional;
import java.util.Set;
import java.util.stream.Collectors;

/**
 * 临时探针（P0c-33 窗口）：打印客户端契约门禁的完整 introduced / repaired 集合（门禁只印前 30 条）。
 * Temporary probe: dump the full introduced/repaired client-contract fingerprint sets.
 */
class RetailContractDeltaProbeTest {

	private static final Path CLIENT_MAPPING = Path.of("docs/quest/client-dialog-mapping");
	private static final String BASELINE_RESOURCE = "/quest/quest-client-contract-baseline.tsv";

	@Test
	void dumpFullContractDelta() throws Exception {
		QuestCatalog catalog = com.aionemu.gameserver.questEngine.retail.RetailQuestDriver.overlay(
			QuestDefinitionDirectoryLoader.compile(getClass().getClassLoader()));
		Map<Integer, QuestDialogOrderAudit.ClientQuest> clientQuests = QuestDialogOrderAudit.readClientPages(
			CLIENT_MAPPING.resolve("quest-dialog-pages.csv"),
			CLIENT_MAPPING.resolve("quest-dialog-action-details.csv"));
		List<QuestDialogOrderAudit.AuditRow> rows = QuestDialogOrderAudit.audit(catalog, clientQuests);
		Set<String> current = rows.stream().map(RetailContractDeltaProbeTest::fatalFingerprint)
			.flatMap(Optional::stream).collect(Collectors.toCollection(LinkedHashSet::new));
		Set<String> baseline = loadBaseline();
		List<String> introduced = new ArrayList<>(current.stream().filter(value -> !baseline.contains(value)).toList());
		introduced.sort(String::compareTo);
		List<String> repaired = new ArrayList<>(baseline.stream().filter(value -> !current.contains(value)).toList());
		repaired.sort(String::compareTo);
		System.out.println("CONTRACT_CURRENT=" + current.size() + " BASELINE=" + baseline.size()
			+ " INTRODUCED=" + introduced.size() + " REPAIRED=" + repaired.size());
		System.out.println("--- INTRODUCED ---");
		introduced.forEach(value -> System.out.println("  " + value.replace('\t', '|')));
		System.out.println("--- REPAIRED ---");
		repaired.forEach(value -> System.out.println("  " + value.replace('\t', '|')));
	}

	private static Optional<String> fatalFingerprint(QuestDialogOrderAudit.AuditRow row) {
		String reason = row.unresolvedReason();
		String failureType;
		if (reason.startsWith("compiled IR emits a task page absent from the active client page index")) {
			failureType = "PAGE_NOT_IN_TASK_HTML";
		} else if (reason.startsWith("visible client action has no route")) {
			failureType = "BUTTON_WITHOUT_ROUTE";
		} else {
			return Optional.empty();
		}
		return Optional.of(String.join("\t", failureType, Integer.toString(row.questId()),
			row.serverSourceState(), row.npcId(), row.triggerAction(), row.shownPage(),
			row.clientVisibleAction(), row.actualPath()));
	}

	private static Set<String> loadBaseline() throws Exception {
		InputStream input = RetailContractDeltaProbeTest.class.getResourceAsStream(BASELINE_RESOURCE);
		Set<String> fingerprints = new LinkedHashSet<>();
		try (BufferedReader reader = new BufferedReader(new InputStreamReader(input, StandardCharsets.UTF_8))) {
			String line;
			while ((line = reader.readLine()) != null) {
				if (line.isBlank() || line.startsWith("#")) {
					continue;
				}
				fingerprints.add(line);
			}
		}
		return fingerprints;
	}
}
