package com.aionemu.gameserver.questEngine.definition;

import com.aionemu.gameserver.questEngine.retail.RetailQuestDriver;

import org.junit.jupiter.api.Test;

import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Map;
import java.util.TreeMap;

/**
 * P0c-54 阶段页轴临时探针（用后即删、源码归档到 {@code .agents/summary/scriptdll-quest-driver/}）。
 * 打印受裁任务的**对话顺序审计行**（{@link QuestDialogOrderAudit} 同口径），用于「入口页补行是否
 * 真的让 CLIENT_PAGE_UNREACHED 闭合 / 剪行是否引入新的未达页」的逐任务机械对拍。
 * Temporary P0c-54 probe: dumps the client-page audit rows for the adjudicated quests.
 */
class P0c57AuditProbeTest {

	@Test
	void dumpAuditRowsForAdjudicatedQuests() throws Exception {
		String out = System.getProperty("p0c57.auditOut");
		String ids = System.getProperty("p0c57.questIds");
		if (out == null || ids == null) {
			return;
		}
		List<Integer> wanted = new ArrayList<>();
		boolean all = ids.trim().equalsIgnoreCase("ALL");
		if (!all) {
			for (String id : ids.split(",")) {
				wanted.add(Integer.parseInt(id.trim()));
			}
		}
		// 生产视图（真端 overlay）：本片受裁任务已退役 XML，目录侧查不到，必须走 overlay（P0c-53 同口径）。
		QuestCatalog catalog = RetailQuestDriver.overlay(
			QuestDefinitionDirectoryLoader.compile(getClass().getClassLoader()));
		List<QuestDialogOrderAudit.AuditRow> rows = QuestDialogOrderAudit.audit(catalog,
			QuestDialogOrderAudit.readClientPages(
				Path.of("docs/quest/client-dialog-mapping/quest-dialog-pages.csv"),
				Path.of("docs/quest/client-dialog-mapping/quest-dialog-action-details.csv")));
		Map<Integer, List<QuestDialogOrderAudit.AuditRow>> byQuest = new TreeMap<>();
		for (QuestDialogOrderAudit.AuditRow row : rows) {
			if (all || wanted.contains(row.questId())) {
				byQuest.computeIfAbsent(row.questId(), _k -> new ArrayList<>()).add(row);
			}
		}
		StringBuilder text = new StringBuilder();
		int interesting = 0;
		for (Map.Entry<Integer, List<QuestDialogOrderAudit.AuditRow>> entry : byQuest.entrySet()) {
			List<QuestDialogOrderAudit.AuditRow> list = new ArrayList<>(entry.getValue());
			list.sort(Comparator.comparing(QuestDialogOrderAudit.AuditRow::auditStatus)
				.thenComparing(QuestDialogOrderAudit.AuditRow::shownPage)
				.thenComparing(QuestDialogOrderAudit.AuditRow::clientVisibleAction));
			text.append("### quest ").append(entry.getKey()).append(" rows=").append(list.size()).append('\n');
			for (QuestDialogOrderAudit.AuditRow row : list) {
				boolean flag = row.auditStatus().equals("CLIENT_PAGE_UNREACHED")
					|| row.auditStatus().equals("EVIDENCE_REQUIRED");
				if (flag) {
					interesting++;
				}
				text.append(flag ? "! " : "  ").append(row.auditStatus()).append('\t')
					.append(row.shownPage()).append('\t').append(row.clientVisibleAction()).append('\t')
					.append(row.actualPath()).append('\t').append(row.unresolvedReason()).append('\n');
			}
		}
		Path path = Path.of(out);
		Files.writeString(path, text.toString());
		System.out.println("P0C57-AUDIT-DUMP quests=" + (all ? "ALL" : wanted.size()) + " withRows=" + byQuest.size()
			+ " unreachedOrEvidence=" + interesting + " out=" + path);
	}
}
