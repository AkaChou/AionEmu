package com.aionemu.gameserver.questEngine.definition;

import com.aionemu.gameserver.questEngine.retail.RetailQuestDriver;

import org.junit.jupiter.api.Test;

import java.nio.file.Path;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.TreeMap;

/**
 * 临时探针（P0c-41）：select6（2716）失败页的关闭出口闭包——全目录扫描"哪些状态下发 select6 页
 * 却没有 1008 关闭路由"，并逐条打印客户端契约致命指纹（修复前/后各跑一轮）。
 * Temporary P0c-41 probe: which states push the select6 page without a FINISH_DIALOG close route.
 */
class P0c41Select6CloseExitProbeTest {

	private static final int SELECT6_PAGE = 2716;
	private static final int FINISH_DIALOG = 1008;

	@Test
	void scanSelect6PushersAndContractFatals() throws Exception {
		QuestCatalog production = RetailQuestDriver.overlay(
			QuestDefinitionDirectoryLoader.compile(getClass().getClassLoader()));
		Map<Integer, QuestDialogOrderAudit.ClientQuest> client = QuestDialogOrderAudit.readClientPages(
			Path.of("docs/quest/client-dialog-mapping/quest-dialog-pages.csv"),
			Path.of("docs/quest/client-dialog-mapping/quest-dialog-action-details.csv"));
		List<QuestDialogOrderAudit.AuditRow> rows = QuestDialogOrderAudit.audit(production, client);

		Map<String, Integer> statuses = new TreeMap<>();
		rows.forEach(row -> statuses.merge(row.auditStatus(), 1, Integer::sum));
		System.out.println("=== P0C41 statuses=" + statuses + " ===");
		List<String> fatal = new ArrayList<>();
		for (QuestDialogOrderAudit.AuditRow row : rows) {
			String reason = row.unresolvedReason();
			if (reason.startsWith("compiled IR emits a task page absent from the active client page index")) {
				fatal.add("PAGE_NOT_IN_TASK_HTML\t" + row.questId() + "\t" + row.shownPage() + "\t"
					+ row.actualPath());
			} else if (reason.startsWith("visible client action has no route")) {
				fatal.add("BUTTON_WITHOUT_ROUTE\t" + row.questId() + "\t" + row.shownPage() + "\t"
					+ row.clientVisibleAction() + "\t" + row.actualPath());
			}
		}
		System.out.println("FATAL count=" + fatal.size());
		fatal.forEach(line -> System.out.println("  " + line));

		System.out.println("=== SELECT6 pushers without close exit ===");
		com.aionemu.gameserver.questEngine.retail.RetailClientDialogExits exits;
		try (java.io.InputStream input = getClass().getResourceAsStream(
				"/aion/data/static_data/quest_retail/quest_client_dialog_exits.tsv")) {
			exits = com.aionemu.gameserver.questEngine.retail.RetailClientDialogExits.load(input);
		}
		int pushers = 0;
		int missing = 0;
		int missingWithExit = 0;
		List<String> scoped = new ArrayList<>();
		for (int questId : new java.util.TreeSet<>(client.keySet())) {
			CompiledQuestDefinition compiled = production.findExecutable(questId).orElse(null);
			if (compiled == null) {
				continue;
			}
			boolean select6Exit = exits.requires(questId,
				com.aionemu.gameserver.questEngine.retail.RetailClientDialogExits.SELECT6);
			QuestDefinition definition = compiled.definition();
			Map<String, List<QuestTransition>> bySource = new LinkedHashMap<>();
			definition.transitions().forEach(t -> bySource
				.computeIfAbsent(String.valueOf(t.sourceNode()), key -> new ArrayList<>()).add(t));
			for (QuestTransition transition : definition.transitions()) {
				boolean pushes = transition.afterCommit().stream()
					.anyMatch(action -> action instanceof AfterCommitAction.ShowQuestDialog dialog
						&& dialog.dialogId() == SELECT6_PAGE);
				if (!pushes || !(transition.event() instanceof QuestEvent.TalkToNpc talk)) {
					continue;
				}
				pushers++;
				boolean closed = bySource.getOrDefault(String.valueOf(transition.sourceNode()), List.of()).stream()
					.anyMatch(candidate -> candidate.event() instanceof QuestEvent.TalkToNpc other
						&& other.npcId() == talk.npcId() && other.dialogId() != null
						&& other.dialogId() == FINISH_DIALOG);
				if (!closed) {
					missing++;
					if (select6Exit) {
						missingWithExit++;
						scoped.add("  SCOPED quest=" + questId + " source=" + transition.sourceNode()
							+ " npc=" + talk.npcId() + " dialogId=" + talk.dialogId()
							+ " clientHasPage=" + client.get(questId).pages().containsKey(SELECT6_PAGE));
					}
				}
			}
		}
		scoped.forEach(System.out::println);
		System.out.println("SELECT6 pushers=" + pushers + " missing close exit=" + missing
			+ " missing WITH select6 exit=" + missingWithExit);
	}

	@Test
	void dumpRows() throws Exception {
		List<Integer> ids = List.of(1932, 3092);
		QuestCatalog production = RetailQuestDriver.overlay(
			QuestDefinitionDirectoryLoader.compile(getClass().getClassLoader()));
		Map<Integer, QuestDialogOrderAudit.ClientQuest> client = QuestDialogOrderAudit.readClientPages(
			Path.of("docs/quest/client-dialog-mapping/quest-dialog-pages.csv"),
			Path.of("docs/quest/client-dialog-mapping/quest-dialog-action-details.csv"));
		List<QuestDialogOrderAudit.AuditRow> rows = QuestDialogOrderAudit.audit(production, client);
		for (int questId : ids) {
			CompiledQuestDefinition compiled = production.findExecutable(questId).orElseThrow();
			System.out.println("=== QUEST " + questId + " IR ===");
			compiled.definition().transitions().stream()
				.sorted(Comparator
					.comparing((QuestTransition t) -> t.sourceNode(),
						Comparator.nullsFirst(Comparator.naturalOrder()))
					.thenComparing(t -> String.valueOf(t.priority()))
					.thenComparing(t -> t.event().toString()))
				.forEach(t -> System.out.println("  T src=" + t.sourceNode() + " " + t.event()
					+ " -> " + t.targetNode() + " pri=" + t.priority() + " cond=" + t.conditions()
					+ " act=" + t.actions() + " after=" + t.afterCommit()));
			QuestDialogOrderAudit.ClientQuest quest = client.get(questId);
			System.out.println("--- CLIENT PAGES " + questId + " ---");
			quest.pages().forEach((page, actions) -> System.out.println("  P " + page + " : " + actions));
			System.out.println("--- AUDIT " + questId + " ---");
			rows.stream().filter(row -> row.questId() == questId)
				.forEach(row -> System.out.println("  A " + row.auditStatus() + " | " + row.shownPage()
					+ " | action=" + row.clientVisibleAction() + " | path=" + row.actualPath()
					+ " | reason=" + row.unresolvedReason()));
		}
	}
}
