package com.aionemu.gameserver.questEngine.definition;

import com.aionemu.gameserver.questEngine.retail.RetailClientTalkChainSteps;
import com.aionemu.gameserver.questEngine.retail.RetailQuestDriver;

import org.junit.jupiter.api.Test;

import java.nio.file.Path;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.TreeMap;

/**
 * 临时探针（P0c-42）：客户端接取入口页 select1(1011) 的"页从未被下发"轴（ACCEPT_PAGE_NOT_EMITTED）。
 * 扫描：客户端页索引里有 1011（且带按钮）的任务，其编译后 IR 是否有任何转换下发 1011；
 * 并对缺口行打印分类事实（族归属 / 链登记 / systemGrant / NPC_START 块的 startPage / 审计行）。
 * Temporary P0c-42 probe: which quests carry an active client select1(1011) page that no transition emits.
 */
class P0c42AcceptPageNotEmittedProbeTest {

	private static final int ACCEPT_ENTRY_PAGE = 1011;

	@Test
	void scanAcceptEntryPageGaps() throws Exception {
		QuestCatalog production = RetailQuestDriver.overlay(
			QuestDefinitionDirectoryLoader.compile(getClass().getClassLoader()));
		Map<Integer, QuestDialogOrderAudit.ClientQuest> client = QuestDialogOrderAudit.readClientPages(
			Path.of("docs/quest/client-dialog-mapping/quest-dialog-pages.csv"),
			Path.of("docs/quest/client-dialog-mapping/quest-dialog-action-details.csv"));
		List<QuestDialogOrderAudit.AuditRow> rows = QuestDialogOrderAudit.audit(production, client);
		RetailClientTalkChainSteps steps;
		try (java.io.InputStream input = getClass().getResourceAsStream(
				"/aion/data/static_data/quest_retail/quest_client_talk_chain_steps.tsv")) {
			steps = RetailClientTalkChainSteps.load(input);
		}

		Map<String, Integer> statuses = new TreeMap<>();
		rows.forEach(row -> statuses.merge(row.auditStatus(), 1, Integer::sum));
		System.out.println("=== P0C42 statuses=" + statuses + " ===");

		// 客户端页索引里的 1011（含按钮才算"活跃页"，审计判据同口径）。
		List<Integer> carriers = client.entrySet().stream()
			.filter(entry -> entry.getValue().pages().containsKey(ACCEPT_ENTRY_PAGE))
			.filter(entry -> production.findExecutable(entry.getKey()).isPresent())
			.sorted(Map.Entry.comparingByKey())
			.map(Map.Entry::getKey)
			.toList();
		System.out.println("client select1(1011) carriers (executable) = " + carriers.size());

		List<String> gaps = new ArrayList<>();
		List<String> emitted = new ArrayList<>();
		Map<String, Integer> gapByFamily = new TreeMap<>();
		int gapChain = 0;
		int gapSystemGrant = 0;
		int gapWithStartBlock = 0;
		for (int questId : carriers) {
			QuestDialogOrderAudit.ClientQuest quest = client.get(questId);
			QuestDialogOrderAudit.ClientPage page = quest.pages().get(ACCEPT_ENTRY_PAGE);
			CompiledQuestDefinition compiled = production.findExecutable(questId).orElseThrow();
			QuestDefinition definition = compiled.definition();
			boolean pushes = definition.transitions().stream()
				.anyMatch(transition -> transition.afterCommit().stream()
					.anyMatch(action -> action instanceof AfterCommitAction.ShowQuestDialog dialog
						&& dialog.dialogId() == ACCEPT_ENTRY_PAGE));
			boolean systemGrant = definition.transitions().stream()
				.anyMatch(transition -> transition.event() instanceof QuestEvent.SystemGrant);
			List<RetailClientTalkChainSteps.BlockRecord> starts = steps.blocks(questId, "NPC_START");
			String startPages = starts.stream()
				.map(block -> block.extra().split("\\|")[1])
				.distinct().sorted().reduce((left, right) -> left + "," + right).orElse("-");
			String owner = manifestOf(questId, 1);
			String family = manifestOf(questId, 2);
			String reason = manifestOf(questId, 3);
			String auditStatus = rows.stream()
				.filter(row -> row.questId() == questId && row.auditStatus().equals("CLIENT_PAGE_UNREACHED")
					&& row.shownPage().equals(Integer.toString(ACCEPT_ENTRY_PAGE)))
				.map(QuestDialogOrderAudit.AuditRow::auditStatus).findFirst().orElse("-");
			String line = "  " + (pushes ? "EMITTED" : "GAP") + " quest=" + questId
				+ " owner=" + owner
				+ " family=" + family
				+ " pageOrder=" + page.pageOrder()
				+ " clientButtons=" + page.actions().keySet()
				+ " chain=" + (!starts.isEmpty() || !steps.nodes(questId).isEmpty())
				+ " systemGrant=" + systemGrant
				+ " npcStartBlocks=" + starts.size()
				+ " startPages=" + startPages
				+ " audit=" + auditStatus
				+ " transitions=" + definition.transitions().size()
				+ " reason=" + reason;
			if (pushes) {
				emitted.add(line);
			} else {
				gaps.add(line);
				gapByFamily.merge(owner + "/" + family + "/order" + page.pageOrder(), 1, Integer::sum);
				if (!starts.isEmpty() || !steps.nodes(questId).isEmpty()) {
					gapChain++;
				}
				if (systemGrant) {
					gapSystemGrant++;
				}
				if (!starts.isEmpty()) {
					gapWithStartBlock++;
				}
			}
		}
		System.out.println("=== ACCEPT ENTRY PAGE GAPS (" + gaps.size() + ") ===");
		gaps.forEach(System.out::println);
		System.out.println("=== ACCEPT ENTRY PAGE EMITTED (" + emitted.size() + ") ===");
		emitted.forEach(System.out::println);
		System.out.println("P0C42 carriers=" + carriers.size() + " gaps=" + gaps.size()
			+ " emitted=" + emitted.size() + " gapChain=" + gapChain
			+ " gapSystemGrant=" + gapSystemGrant + " gapWithStartBlock=" + gapWithStartBlock
			+ " gapByFamily=" + gapByFamily);

		// 全目录 1011 未达页（不问客户端页集，直接看审计判据）
		Map<String, Integer> unreached1011 = new LinkedHashMap<>();
		rows.stream().filter(row -> row.auditStatus().equals("CLIENT_PAGE_UNREACHED")
				&& row.shownPage().equals(Integer.toString(ACCEPT_ENTRY_PAGE)))
			.forEach(row -> unreached1011.merge(Integer.toString(row.questId()), 1, Integer::sum));
		System.out.println("P0C42 unresolved-1011 quests=" + unreached1011.size());
	}

	@Test
	void dumpQuest() throws Exception {
		List<Integer> ids = List.of(1323);
		QuestCatalog production = RetailQuestDriver.overlay(
			QuestDefinitionDirectoryLoader.compile(getClass().getClassLoader()));
		Map<Integer, QuestDialogOrderAudit.ClientQuest> client = QuestDialogOrderAudit.readClientPages(
			Path.of("docs/quest/client-dialog-mapping/quest-dialog-pages.csv"),
			Path.of("docs/quest/client-dialog-mapping/quest-dialog-action-details.csv"));
		List<QuestDialogOrderAudit.AuditRow> rows = QuestDialogOrderAudit.audit(production, client);
		for (int questId : ids) {
			CompiledQuestDefinition compiled = production.findExecutable(questId).orElseThrow();
			System.out.println("=== QUEST " + questId + " IR ===");
			compiled.definition().transitions().stream()
				.sorted(Comparator
					.comparing((QuestTransition t) -> t.sourceNode(),
						Comparator.nullsFirst(Comparator.naturalOrder()))
					.thenComparing(t -> String.valueOf(t.priority()))
					.thenComparing(t -> t.event().toString()))
				.forEach(t -> System.out.println("  T src=" + t.sourceNode() + " " + t.event()
					+ " -> " + t.targetNode() + " pri=" + t.priority() + " cond=" + t.conditions()
					+ " act=" + t.actions() + " after=" + t.afterCommit()));
			QuestDialogOrderAudit.ClientQuest quest = client.get(questId);
			System.out.println("--- CLIENT PAGES " + questId + " ---");
			quest.pages().entrySet().stream()
				.sorted(Map.Entry.comparingByKey())
				.forEach(entry -> System.out.println("  P " + entry.getKey()
					+ " order=" + entry.getValue().pageOrder()
					+ " name=" + entry.getValue().pageName()
					+ " actions=" + entry.getValue().actions().keySet()
					+ " unmapped=" + entry.getValue().unmappedActions().keySet()));
			System.out.println("--- AUDIT " + questId + " ---");
			rows.stream().filter(row -> row.questId() == questId)
				.forEach(row -> System.out.println("  A " + row.auditStatus() + " | page=" + row.shownPage()
					+ " | action=" + row.clientVisibleAction() + " | path=" + row.actualPath()
					+ " | reason=" + row.unresolvedReason()));
		}
	}

	/** retention 清单列（owner=1 / family=2 / reason=3）。 / Retention manifest column lookup. */
	private static String manifestOf(int questId, int column) {
		try (java.io.InputStream input = P0c42AcceptPageNotEmittedProbeTest.class.getResourceAsStream(
				"/aion/data/static_data/quest_retail/retail-xml-retention.tsv")) {
			if (input == null) {
				return "?";
			}
			for (String line : new String(input.readAllBytes(), java.nio.charset.StandardCharsets.UTF_8)
					.split("\n")) {
				String[] parts = line.split("\t", -1);
				if (parts.length > column && parts[0].equals(Integer.toString(questId))) {
					return parts[column];
				}
			}
		} catch (java.io.IOException ignored) {
			return "?";
		}
		return "?";
	}

	/** 未使用，保留探针调试入口。 / Unused helper kept for probe debugging. */
	static List<Integer> sortedIds(java.util.Collection<Integer> ids) {
		List<Integer> sorted = new ArrayList<>(ids);
		sorted.sort(Comparator.naturalOrder());
		return sorted;
	}
}
