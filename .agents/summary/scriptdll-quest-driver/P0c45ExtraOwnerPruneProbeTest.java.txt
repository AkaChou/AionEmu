package com.aionemu.gameserver.questEngine.definition;

import com.aionemu.gameserver.questEngine.retail.RetailClientTalkChainSteps;
import com.aionemu.gameserver.questEngine.retail.RetailQuestDriver;

import org.junit.jupiter.api.Test;

import java.io.InputStream;
import java.nio.file.Path;
import java.util.Comparator;
import java.util.List;
import java.util.Map;

/**
 * 临时探针（P0c-45）：35010/35011 剪除多余交付 owner（799806）后的审计逐行 + 路由详情。
 * 证据用途：证明"删掉多余 owner 后没有任何客户端页变成未达"（修复面 = 缺陷面）。
 * Temporary P0c-45 probe: per-quest audit rows + route detail for the two pruned quests.
 */
class P0c45ExtraOwnerPruneProbeTest {

	private static final int[] QUESTS = {35010, 35011};

	@Test
	void dumpQuests() throws Exception {
		QuestCatalog production = RetailQuestDriver.overlay(
			QuestDefinitionDirectoryLoader.compile(getClass().getClassLoader()));
		Map<Integer, QuestDialogOrderAudit.ClientQuest> client = QuestDialogOrderAudit.readClientPages(
			Path.of("docs/quest/client-dialog-mapping/quest-dialog-pages.csv"),
			Path.of("docs/quest/client-dialog-mapping/quest-dialog-action-details.csv"));
		List<QuestDialogOrderAudit.AuditRow> rows = QuestDialogOrderAudit.audit(production, client);
		RetailClientTalkChainSteps steps;
		try (InputStream input = getClass().getResourceAsStream(
				"/aion/data/static_data/quest_retail/quest_client_talk_chain_steps.tsv")) {
			steps = RetailClientTalkChainSteps.load(input);
		}
		for (int questId : QUESTS) {
			CompiledQuestDefinition compiled = production.findExecutable(questId).orElseThrow();
			QuestDefinition definition = compiled.definition();
			System.out.println("--- P0C45 AUDIT " + questId + " (status\tpage\taction\tpath\treason) ---");
			rows.stream().filter(row -> row.questId() == questId)
				.forEach(row -> System.out.println("  A " + String.join("\t",
					String.valueOf(row.auditStatus()), row.shownPage(), row.clientVisibleAction(),
					row.actualPath(), String.valueOf(row.unresolvedReason()))));
			System.out.println("--- P0C45 TRANSITIONS " + questId + " ---");
			definition.transitions().stream()
				.sorted(Comparator
					.comparing((QuestTransition t) -> String.valueOf(t.sourceNode()))
					.thenComparing(t -> String.valueOf(t.priority()))
					.thenComparing(t -> t.event().toString()))
				.forEach(t -> System.out.println("  T " + t.sourceNode() + " " + t.event()
					+ " -> " + t.targetNode() + " pri=" + t.priority()));
			System.out.println("--- P0C45 BLOCKS " + questId + " ---");
			for (String block : List.of("NPC_COMPLETE")) {
				steps.blocks(questId, block).forEach(record -> System.out.println("  B " + record));
			}
			long unreached = rows.stream().filter(row -> row.questId() == questId)
				.filter(row -> String.valueOf(row.auditStatus()).contains("UNREACHED")).count();
			long extraOwner = definition.transitions().stream()
				.filter(t -> t.event().toString().contains("799806")).count();
			System.out.println("P0C45 quest=" + questId
				+ " auditRows=" + rows.stream().filter(row -> row.questId() == questId).count()
				+ " unreached=" + unreached
				+ " transitions=" + definition.transitions().size()
				+ " refsTo799806=" + extraOwner
				+ " completeBlocks=" + steps.blocks(questId, "NPC_COMPLETE").size());
		}
	}
}
