package com.aionemu.gameserver.questEngine.definition;

import org.junit.jupiter.api.Test;

/**
 * 临时探针（P0c-27）：1351/1526 真端编译形状直证（lenient 视图，风暴期可用）。
 * 1351——接取 NPC 203965 的 started 出口族形（预期仅 FINISH_DIALOG，无 QUEST_SELECT→SELECT1）；
 * 1526——reward 投影 var0（预期 = 2，登记表修正后）。
 * Temporary probe: dump 1351's started exits on NPC 203965 (expect FINISH_DIALOG-only family
 * shape) and 1526's reward projection (expect var0=2 after the summary-rows fix).
 */
class Retail1526And1351OverlayProbeTest {

	@Test
	void dump1351StartedExitsAnd1526RewardProjection() {
		QuestDefinition d1351 = ProductionQuestDefinitions.definitionInOverlay(1351).definition();
		System.out.println("=== 1351 started exits on 203965 ===");
		d1351.transitions().stream()
			.filter(transition -> "started".equals(transition.sourceNode()))
			.filter(transition -> transition.event() instanceof QuestEvent.TalkToNpc talk
				&& talk.npcId() == 203965)
			.forEach(transition -> System.out.println("EXIT " + transition.event()
				+ " cond=" + transition.conditions() + " act=" + transition.actions()
				+ " target=" + transition.targetNode() + " prio=" + transition.priority()
				+ " after=" + transition.afterCommit()));
		System.out.println("=== 1351 turn-in NPC (reward ownership) exits ===");
		d1351.transitions().stream()
			.filter(transition -> "started".equals(transition.sourceNode())
				&& "reward".equals(transition.targetNode()))
			.forEach(transition -> System.out.println("TURNIN " + transition.event()
				+ " cond=" + transition.conditions() + " act=" + transition.actions()));
		System.out.println("=== 1351 reward projection ===");
		System.out.println("1351 reward vars: " + d1351.nodes().stream()
			.filter(node -> node.label().equals("reward")).findFirst().orElseThrow()
			.projection().variables());

		QuestDefinition d1526 = ProductionQuestDefinitions.definitionInOverlay(1526).definition();
		System.out.println("=== 1526 reward projection ===");
		System.out.println("1526 reward vars: " + d1526.nodes().stream()
			.filter(node -> node.label().equals("reward")).findFirst().orElseThrow()
			.projection().variables());
	}
}
