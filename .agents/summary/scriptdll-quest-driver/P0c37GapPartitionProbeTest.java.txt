package com.aionemu.gameserver.questEngine.definition;

import com.aionemu.gameserver.questEngine.retail.RetailQuestDriver;

import org.junit.jupiter.api.Test;

import java.nio.file.Path;
import java.util.List;
import java.util.Map;
import java.util.TreeSet;

/**
 * 临时探针（P0c-37）：把 P0c-36 普查的"缺阶梯"判定与**客户端页可达性**对拍，区分两类形状：
 * ①真缺口 = 中间对话页（SELECT 前缀页与 SETPRO 页）在编译后的 IR 里 `CLIENT_PAGE_UNREACHED`（无任何路由）；
 * ②旗标形 = 阶段以单 START 节点上的 var0 旗标 + 条件化路由承载（中间页已 `PAGE_ACTION_MATCHED`），
 *   属普查口径误报（节点投影轴看不见 transition 级编码），不应重写。
 * Temporary probe: cross-check the "missing ladder" census against client-page reachability.
 */
class P0c37GapPartitionProbeTest {

	/** P0c-36 普查的 39 行余量（含已裁定的 24202/80320 作为对照）。 */
	private static final int[] GAP_ROWS = {1324, 1463, 1471, 1479, 2428, 2963, 11001, 13800, 18035, 18600,
		18806, 18807, 18809, 19064, 21070, 21138, 21244, 21460, 23800, 24120, 24123, 24202, 28035,
		28600, 28806, 28807, 29064, 29070, 29071, 30055, 30202, 30302, 30711, 30761, 35010, 35018,
		35024, 45011, 45025, 80021, 80320};

	@Test
	void partitionGapRows() throws Exception {
		QuestCatalog catalog = RetailQuestDriver.overlay(
			QuestDefinitionDirectoryLoader.compile(getClass().getClassLoader()));
		Map<Integer, QuestDialogOrderAudit.ClientQuest> clientQuests = QuestDialogOrderAudit.readClientPages(
			Path.of("docs/quest/client-dialog-mapping/quest-dialog-pages.csv"),
			Path.of("docs/quest/client-dialog-mapping/quest-dialog-action-details.csv"));
		List<QuestDialogOrderAudit.AuditRow> rows = QuestDialogOrderAudit.audit(catalog, clientQuests);
		int genuine = 0;
		int flagged = 0;
		int noDefinition = 0;
		for (int questId : GAP_ROWS) {
			List<QuestDialogOrderAudit.AuditRow> mine = rows.stream()
				.filter(row -> row.questId() == questId).toList();
			boolean executable = catalog.findExecutable(questId).isPresent();
			TreeSet<String> unreachedMiddle = new TreeSet<>();
			TreeSet<String> unreachedOther = new TreeSet<>();
			for (QuestDialogOrderAudit.AuditRow row : mine) {
				if (!"CLIENT_PAGE_UNREACHED".equals(row.auditStatus())) {
					continue;
				}
				String page = String.valueOf(row.shownPage());
				QuestDialogOrderAudit.ClientQuest clientQuest = clientQuests.get(questId);
				QuestDialogOrderAudit.ClientPage clientPage = null;
				if (clientQuest != null && page.chars().allMatch(Character::isDigit)) {
					clientPage = clientQuest.pages().get(Integer.valueOf(page));
				}
				String name = clientPage == null ? null : clientPage.pageName();
				String label = page + "/" + (name == null ? "?" : name);
				String upper = name == null ? "" : name.toUpperCase();
				if (upper.startsWith("SELECT") || upper.startsWith("QUEST_SUMMARY")) {
					unreachedMiddle.add(label);
				} else {
					unreachedOther.add(label);
				}
			}
			String verdict;
			if (!executable) {
				verdict = "NO_DEFINITION";
				noDefinition++;
			} else if (!unreachedMiddle.isEmpty()) {
				verdict = "GENUINE_GAP";
				genuine++;
			} else {
				verdict = "CENSUS_OVERFLAG";
				flagged++;
			}
			System.out.println("PART " + questId + " executable=" + executable + " " + verdict
				+ " unreachedMiddle=" + unreachedMiddle + " unreachedOther=" + unreachedOther);
		}
		System.out.println("PARTITION genuine=" + genuine + " overflag=" + flagged
			+ " noDefinition=" + noDefinition + " total=" + GAP_ROWS.length);
	}
}
