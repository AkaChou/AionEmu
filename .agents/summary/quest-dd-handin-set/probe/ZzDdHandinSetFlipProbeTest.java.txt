package com.aionemu.gameserver.questEngine.definition;

import org.junit.jupiter.api.Test;

import java.nio.file.Files;
import java.nio.file.Path;

/** 临时探针：转储 DD 领奖集合轴解锁行的生产 IR（用完即删）。 / Temporary IR dump probe. */
class ZzDdHandinSetFlipProbeTest {

	private static final int[] QUESTS = {30722, 30772, 50064, 50108, 51064};

	@Test
	void dump() throws Exception {
		String out = System.getProperty("flip.probe.out");
		StringBuilder sb = new StringBuilder();
		for (int questId : QUESTS) {
			QuestDefinition definition = ProductionQuestDefinitions.definitionInOverlay(questId).definition();
			sb.append("==================== QUEST ").append(questId).append('\n');
			sb.append("-- nodes\n");
			for (QuestNode node : definition.nodes()) {
				sb.append("   [").append(node.label()).append("] ")
					.append(node.projection().status()).append(' ')
					.append(node.projection().variables()).append('\n');
			}
			sb.append("-- transitions\n");
			for (QuestTransition transition : definition.transitions()) {
				sb.append("   ").append(transition.sourceNode()).append(" --")
					.append(transition.event()).append(" conds=").append(transition.conditions())
					.append(" actions=").append(transition.actions())
					.append(" after=").append(transition.afterCommit())
					.append(" --> ").append(transition.targetNode()).append('\n');
			}
		}
		Files.writeString(Path.of(out), sb.toString());
	}
}
