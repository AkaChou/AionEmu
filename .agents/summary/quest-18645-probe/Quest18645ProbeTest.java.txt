class Quest18645ProbeTest {

	@Test
	void probe() {
		var table = NativeQuestXmlTable.instance();
		var row = table.find(18645);
		System.out.println("PROBE1 row18645=" + row.isPresent());
		row.ifPresent(r -> System.out.println("PROBE2 class=" + r.text("class_permitted")
			+ " | race=" + r.text("race_permitted")
			+ " | min=" + r.text("minlevel_permitted") + " | max=" + r.text("maxlevel_permitted")
			+ " | reward=" + r.text("reward_item1_1")));
		System.out.println("PROBE3 routes(18645)=" + SimpleTalkHandler.instance().routes(18645));
		System.out.println("PROBE4 owns(18645)=" + SimpleTalkHandler.instance().owns(18645));
		System.out.println("PROBE5 acquireNpcs(18645)=" + SimpleTalkHandler.instance().acquireNpcs(18645));
		System.out.println("PROBE6 permitted=" + RetailQuestMetadataCompiler
			.permittedClassNames("warrior scout mage cleric engineer artist bard", 37));
		Player kk = player(48, PlayerClass.SONGWEAVER);
		System.out.println("PROBE7 zoneVerdict(bard48)=" + NativeQuestStartPort.instance().zoneVerdict(kk, 18645));
		System.out.println("PROBE8 evaluateNpcAcquire(bard48)="
			+ NativeQuestStartPort.instance().evaluateNpcAcquire(kk, 18645));
	}

	private static Player player(int level, PlayerClass playerClass) {
		Player player = new ObjenesisStd().newInstance(Player.class);
		PlayerCommonData pcd = new PlayerCommonData(10001);
		pcd.setRace(Race.ELYOS);
		pcd.setGender(Gender.MALE);
		// setPlayerClass / setLevel 需要经验表就绪（单测无服务栈）⇒ 直接写字段。
		setField(pcd, PlayerCommonData.class, "playerClass", playerClass);
		setField(pcd, PlayerCommonData.class, "level", level);
		setField(player, Player.class, "playerCommonData", pcd);
		player.setQuestStateList(new QuestStateList());
		return player;
	}

	private static void setField(Object target, Class<?> type, String name, Object value) {
		try {
			Field field = type.getDeclaredField(name);
			field.setAccessible(true);
			field.set(target, value);
		} catch (ReflectiveOperationException e) {
			throw new RuntimeException(e);
		}
	}
}
